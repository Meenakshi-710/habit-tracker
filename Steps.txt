"use client";

import { useEffect, useState } from "react";
import { FaCalendarAlt, FaRegClock } from "react-icons/fa";
import { initTokenClient, requestAccessToken } from "../lib/googleAuth";

interface StepTwoProps {
  onBack: () => void;
  onNext: () => void;
}

const StepTwo = ({ onBack, onNext }: StepTwoProps) => {
  // eslint-disable-next-line @typescript-eslint/no-unused-vars
  const [, setAccessToken] = useState<string | null>(null);
  const [isConnected, setIsConnected] = useState(false);
  const [showDialog, setShowDialog] = useState(true);
  const [isLoading, setIsLoading] = useState(false);

  // Load GIS script
  useEffect(() => {
    const script = document.createElement("script");
    script.src = "https://accounts.google.com/gsi/client";
    script.async = true;
    script.onload = () => {
      console.log("GIS loaded");
      initTokenClient((token) => {
        setAccessToken(token);
        setIsConnected(true);
        setShowDialog(false);
        localStorage.setItem("google_access_token", token);
      });
    };
    document.body.appendChild(script);

    // Load token from localStorage (if already authorized)
    const savedToken = localStorage.getItem("google_access_token");
    if (savedToken) {
      setAccessToken(savedToken);
      setIsConnected(true);
      setShowDialog(false);
    }
  }, []);

  const connectCalendar = async () => {
    setIsLoading(true);
    try {
      requestAccessToken();
    } catch (error) {
      console.error("Calendar connection failed:", error);
    } finally {
      setIsLoading(false);
    }
  };

  const handleNext = () => {
    onNext();
  };

  return (
    <div className="min-h-screen flex flex-col px-6 md:px-24 pt-6 pb-12 bg-gradient-to-br from-white to-gray-100 dark:from-black dark:to-gray-900 text-black dark:text-white">
      {/* Alert Dialog */}
      {showDialog && (
        <div className="fixed inset-0 bg-black bg-opacity-40 flex items-center justify-center z-50">
          <div className="bg-white dark:bg-gray-800 p-6 rounded-lg shadow-lg w-full max-w-md">
            <h2 className="text-xl font-bold mb-4">Connect Google Calendar</h2>
            <p className="text-gray-600 dark:text-gray-300 mb-4">
              We need access to your calendar to sync your events with your
              habits.
            </p>
            <div className="flex justify-end space-x-3">
              <button
                onClick={() => setShowDialog(false)}
                className="px-4 py-2 text-sm rounded-md bg-gray-200 dark:bg-gray-700 text-black dark:text-white"
              >
                Maybe Later
              </button>
              <button
                onClick={connectCalendar}
                disabled={isLoading}
                className="px-4 py-2 text-sm rounded-md bg-black dark:bg-white text-white dark:text-black font-medium disabled:opacity-50"
              >
                {isLoading ? "Connecting..." : "Connect"}
              </button>
            </div>
          </div>
        </div>
      )}

      {/* Top Navigation */}
      <div className="flex justify-between items-center text-sm text-gray-500 dark:text-gray-400 mb-8">
        <button onClick={onBack} className="hover:underline">
          ← Back
        </button>
        <button onClick={onNext} className="hover:underline">
          Skip
        </button>
      </div>

      {/* Main Content */}
      <div className="flex-grow flex flex-col items-start max-w-2xl mx-auto">
        <p className="text-lg text-gray-500 dark:text-gray-400 mb-2">Step 2</p>

        <h1 className="text-[35px] font-semibold mb-6">
          Connect your <FaCalendarAlt className="inline text-[35px] mx-1" />{" "}
          calendar <br />
          to keep all <FaRegClock className="inline text-[35px] mx-1" /> events
          in one place
        </h1>

        {isConnected && (
          <div className="w-full mb-6 p-4 bg-green-100 dark:bg-green-900 text-green-800 dark:text-green-200 rounded-lg">
            <div className="flex items-center justify-between">
              <span>✅ Calendar connected successfully!</span>
              <button
                onClick={() => {
                  localStorage.removeItem("google_access_token");
                  setAccessToken(null);
                  setIsConnected(false);
                  setShowDialog(true);
                }}
                className="ml-4 px-3 py-1 bg-black text-white rounded-md hover:bg-red-700 transition text-sm"
              >
                X
              </button>
            </div>
          </div>
        )}

        <button
          onClick={handleNext}
          className={`mt-6 w-full text-lg font-semibold py-3 rounded-xl transition ${
            isConnected
              ? "bg-black dark:bg-white text-white dark:text-black hover:opacity-90"
              : "bg-black dark:bg-white text-white hover:bg-black dark:hover:bg-black"
          }`}
        >
          {isConnected ? "Continue to Habit Tracker" : "Skip & Continue"}
        </button>
      </div>
    </div>
  );
};

export default StepTwo;
